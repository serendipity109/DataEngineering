docker network create MBTANetwork

docker run -p 27017:27017 --name some-mongo --network MBTANetwork  -d mongo 

docker run --name javamaven --network MBTANetwork  -dti --rm -p 8080:8080 maven:3.9.9-eclipse-temurin-21 bash

git clone https://github.com/mongodb-developer/java-quick-start
docker cp java-quick-start javamaven:/

apt update
apt install nano -y

cd /java-quick-start/src/main/java/com/mongodb/quickstart
touch ReadCDC.java
nano ReadCDC.java

# nano 語法
ctrl O -> enter 存檔
ctrl X 離開

mvn compile exec:java -Dexec.mainClass="com.mongodb.quickstart.ReadCDC" -Dmongodb.uri="mongodb://some-mongo:27017"
